Sketched generated Python helper (adapt to the existing style in `function-declarations.ts`):
```python
def set_hand_position_xyz(side: str, mode: str, x, y, z):
    if mode == "RELATIVE":
        current = get_current_hand_position(side)   # resolve from pib/sdk
        target = [current[0]+x, current[1]+y, current[2]+z]
    else:
        target = [x, y, z]
    q_deg = ik(side, xyz=target)
    arm = left_arm if side == "left" else right_arm
    with Write(host=rosbridge_host, port=9090) as pib:
        pib.move(arm, *q_deg)
```
